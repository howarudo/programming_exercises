# ABC 164 D - Multiple of 2019

[problem link](https://atcoder.jp/contests/abc164/tasks/abc164_d)


```python
import sys
from collections import Counter, defaultdict, deque

input = sys.stdin.readline
sys.setrecursionlimit(10**6)


def ii():
    return int(input())


def mi():
    return map(int, input().split())


def li():
    return list(map(int, input().split()))


def si():
    return input().strip()


def main():
    S = si()
    n = len(S)
    cur_rem = 0
    _map = defaultdict(int)
    _map[0] = 1
    ans = 0
    pow_10_mod_2019 = 1
    for i in range(n-1, -1, -1):
        num = ord(S[i]) - ord('0')
        cur_rem += (pow_10_mod_2019 * num % 2019)
        pow_10_mod_2019 *= 10
        pow_10_mod_2019 %= 2019
        cur_rem %= 2019
        if cur_rem in _map:
            ans += _map[cur_rem]
        _map[cur_rem] += 1
    print(ans)
    return



if __name__ == "__main__":
    main()

```
