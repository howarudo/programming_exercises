[problem link](https://atcoder.jp/contests/abc159/tasks/abc159_d)


```python
import sys
from collections import Counter, defaultdict, deque

input = sys.stdin.readline
sys.setrecursionlimit(10**6)


def ii():
    return int(input())


def mi():
    return map(int, input().split())


def li():
    return list(map(int, input().split()))


def si():
    return input().strip()


def main():
    N = ii()
    As = li()
    c = Counter(As)
    ans = 0
    for k, v in c.items():
        ans += v * (v - 1) // 2
    for A in As:
        print(ans - c[A] + 1)

    return



if __name__ == "__main__":
    main()
```
