# ABC 218 D - Rectangles

[problem link](https://atcoder.jp/contests/abc218/tasks/abc218_d)


```python
import sys
from collections import Counter, defaultdict, deque

input = sys.stdin.readline
sys.setrecursionlimit(10**6)


def ii():
    return int(input())


def mi():
    return map(int, input().split())


def li():
    return list(map(int, input().split()))


def si():
    return input().strip()


def main():
    N = ii()
    ans = 0
    memo = set()
    for _ in range(N):
        XY = mi()
        X, Y = XY
        for _x, _y in memo:
            if (X, _y) in memo and (_x, Y) in memo:
                ans += 1
        memo.add((X, Y))
    print(ans)
    return



if __name__ == "__main__":
    main()

```
