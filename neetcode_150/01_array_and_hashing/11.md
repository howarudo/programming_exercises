# ABC 146 E - Rem of Sum is Num

[problem link](https://atcoder.jp/contests/abc146/tasks/abc146_e)


```python
import sys
from collections import Counter, defaultdict, deque

input = sys.stdin.readline
sys.setrecursionlimit(10**6)


def ii():
    return int(input())


def mi():
    return map(int, input().split())


def li():
    return list(map(int, input().split()))


def si():
    return input().strip()


def main():
    NK = li()
    N = NK[0]
    K = NK[1]

    A = li()
    _map = defaultdict(int)
    back_sum = 0
    cur_sum = 0
    _map[1] = 1
    ans = 0
    for i in range(N):
        if i == K - 1:
            _map[1] -= 1
        if i >= K:
            back_sum += A[i-K]
            _map[(back_sum - i + K) % K] -= 1
        cur_sum += A[i]
        _mod = ((cur_sum - i) % K)
        if _mod in _map:
            ans += _map[_mod]
        _map[_mod] += 1
    print(ans)
    return



if __name__ == "__main__":
    main()
```
